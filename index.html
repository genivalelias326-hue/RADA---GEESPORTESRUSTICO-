async function requestPlan(plan) { const status = document.getElementById('planStatus'); status.style.display = 'block'; status.textContent = 'Preparando seu pagamento...'; try { const { data: sessionData } = await supabase.auth.getSession(); const session = sessionData.session; if (!session) { status.textContent = 'Faça login para continuar com a assinatura.'; return; } const response = await fetch( 'https://uwdhjiuufrsjqcbgtoyn.supabase.co/functions/v1/rapid-endpoint', { method: 'POST', headers: { 'Content-Type': 'application/json', 'Authorization': 'Bearer ' + session.access_token, 'apikey': SUPABASE_ANON_KEY }, body: JSON.stringify({ plano: plan, user_id: session.user.id, email: session.user.email }) } ); const result = await response.json(); if (!response.ok) { throw new Error(result.message || result.error || 'Não foi possível gerar o pagamento.'); } const checkoutUrl = result.checkoutUrl || result.checkout_url || result.invoiceUrl || result.invoice_url || result.paymentLink || result.payment_link; if (checkoutUrl) { status.textContent = 'Redirecionando para o pagamento...'; window.location.href = checkoutUrl; return; } status.textContent = result.message || 'O servidor respondeu, mas não retornou o link de pagamento.'; } catch (error) { console.error('Erro no pagamento:', error); status.textContent = 'Não foi possível iniciar o pagamento: ' + error.message; } }
